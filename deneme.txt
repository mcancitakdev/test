private async Task InitializeMachineConnectionAsync()
{
    if (_connectionScanRunning)
        return;

    await _connectionScanLock.WaitAsync();

    try
    {
        if (_connectionScanRunning)
            return;

        _connectionScanRunning = true;
        _connected = false;

        btn_connect.Enabled = false;
        lblComPortState.Text = "State : Searching machine...";

        SetLogWriteLine(
            $"[{DateTime.Now:yyyy-MM-dd HH:mm:ss}] " +
            "Machine search started.");

        if (!Protocol.CheckDll())
        {
            lblComPortState.Text = "State : Protocol DLL invalid";
            MessageBox.Show("Protocol DLL reference is invalid!");
            return;
        }

        int portType = GetSelectedPortType();

        MachineConnectionResult machine =
            await FindAndConnectMachineAsync(portType);

        if (machine == null)
        {
            _connected = false;
            _model = Models.Unknown;
            _strMachineSN = string.Empty;

            lblComPortState.Text = "State : Machine not found";

            SetLogWriteLine(
                $"[{DateTime.Now:yyyy-MM-dd HH:mm:ss}] " +
                "Supported machine could not be found.");

            return;
        }

        // Port halen acik ve MachineInfo cevabi dogrulandi.
        _model = machine.Model;
        _strMachineSN = machine.MachineSerialNumber;
        _connected = true;

        lblComPortState.Text =
            $"State : {machine.ModelName} Connected - " +
            $"{machine.PortName} / {machine.BaudRate}";

        SelectComboBoxItem(cbb_portName, machine.PortName);
        SelectComboBoxItem(cbb_baudRate, machine.BaudRate);

        SetLogWriteLine(
            $"[{DateTime.Now:yyyy-MM-dd HH:mm:ss}] " +
            "Machine connection verified.\r\n" +
            $"  - Port     : {machine.PortName}\r\n" +
            $"  - BaudRate : {machine.BaudRate}\r\n" +
            $"  - Model    : {machine.ModelName}\r\n" +
            $"  - SN       : {machine.MachineSerialNumber}");
    }
    catch (Exception ex)
    {
        _connected = false;

        try
        {
            m_ProtocolDll.CloseSerialPort();
        }
        catch
        {
            // Port zaten kapali olabilir.
        }

        lblComPortState.Text = "State : Connection error";

        SetLogWriteLine(
            $"[{DateTime.Now:yyyy-MM-dd HH:mm:ss}] " +
            "Machine initialization error: " + ex.Message);
    }
    finally
    {
        _connectionScanRunning = false;
        btn_connect.Enabled = true;
        _connectionScanLock.Release();
    }
}
